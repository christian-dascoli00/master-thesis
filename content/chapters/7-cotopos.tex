\chapter{Cotopoi as a fundamental example}

\fontsize{12}{14.5}\selectfont

\section{The general case}

In this section, we aim to show that any cotopos $\mathcal{E}^{\mathrm{op}}$, the dual of a topos $\mathcal{E}$, is an ideally exact category. Then, since the category of $\mathsf{Set}$ is a topos, we will analyze the case of $\mathsf{Set}^{\mathrm{op}}$. This section is based on \cite{Bourn2004}, \cite{Johnstone1977}, \cite{Borceux1994II} and \cite{Borceux1994III}.

First, we recall the definition of a topos.

\begin{mainbox}{}
\begin{defi}
Let $\mathcal{E}$ be a finitely complete category. A \textbf{subobject classifier} of $\mathcal{E}$ is a pair $(\Omega, t)$, where $\Omega$ is an object of $\mathcal{E}$ and $t: 1 \rightarrowtail \Omega$ is a monomorphism, such that for any object $A$ of $\mathcal{E}$ and any monomorphism $m: S \rightarrowtail A$, there exists a unique morphism $\chi_S: A \rightarrow \Omega$ making the diagram
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAGUQBfU9TXfIRQBGclVqMWbADrSAtnRwALAEYAzYMK7deIDNjwEiZYePrNWiEAEEdfA4KKjT1c1KuyA8nJgBzOtziMFC+8ESgagBOEHJIZCA4EEhaulExcdSJSABMmXRYDGw4+YWukpYgsdQMdCowDAAK-IZCIJFYvko4diBpsYi5CUmIAMxlFjLSAMZKWAD6nDwR0f2iQ0hjEhNW3VwUXEA
\begin{tikzcd}
S \arrow[r] \arrow[d, "m"', tail] & \mathsf{1} \arrow[d, "t"] \\
A \arrow[r, "\chi_S"]             & \Omega                   
\end{tikzcd}
\]
a pullback.
\end{defi}
\end{mainbox}

\newpage

\begin{mainbox}{}
\begin{defi}
A category $\mathcal{E}$ is said to be an \textbf{elementary topos}, in the following just a \textbf{topos}, if
\begin{itemize}
    \item it is finitely complete;
    \item it is cartesian closed;
    \item it has a subobject classifier.
\end{itemize}
\end{defi}
\end{mainbox}

A topos turns out to have a rich structure. First, we need some properties. Recall that the exponential is given by the cartesian closedness. For all objects $A$, $B$, and $C$ in $\mathcal{E}$, it holds that
\[
\mathrm{Hom}_{\mathcal{E}}(A, \Omega^B) \cong \mathrm{Hom}_{\mathcal{E}}(A \times B, \Omega) \cong \mathrm{Hom}_{\mathcal{E}}(B, \Omega^A)
\]

\begin{mainbox}{}
\begin{prop}\label{exp in topos is monadic}
Given a topos, the exponential functor 
\[\Omega^{(-)}: \mathcal{E}^{\mathrm{op}} \to \mathcal{E} \]
is monadic.
\end{prop}
\end{mainbox}
\noindent The proof is contained in \cite{Borceux1994III} \textit{Theorem 5.7.3} and it is based on the Beck monadicity theorem.

\begin{mainbox}{}
\begin{prop}\label{topos is finitely cocomplete}
A topos is finitely cocomplete.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
Consider a topos $\mathcal{E}$. Since the functor $\Omega^{(-)}: \mathcal{E}^{\mathrm{op}} \to \mathcal{E}$, by \Cref{exp in topos is monadic}, is monadic, and $\mathcal{E}$ is finitely complete, it follows that $\mathcal{E}^{\mathrm{op}}$ is finitely complete. Thus, $\mathcal{E}$ is finitely cocomplete.
\end{proof}
\end{subbox}

\begin{mainbox}{}
\begin{prop}\label{conditions for factorization regular epi mono}
Let $\mathcal{E}$ be a category with the following conditions:
\begin{itemize}
\item it has pullbacks;
\item it has coequalizers of kernel pairs;
\item it has pullback stable regular epimorphisms.
\end{itemize}
Then each morphism $f: A \to B$ in $\mathcal{E}$ admits a (regular epi, mono) factorization.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}

\begin{mainbox}{}
\begin{prop}\label{colimits are universal in topos}
    In a topos, all finite colimits in the slice category are preserved by the pullback functor associated to any morphism.
\end{prop}
\end{mainbox}
\noindent The proof is contained in \cite{Borceux1994II} \textit{Proposition 5.9.1}.

\begin{mainbox}{}

        \item $q$ is pullback stable;

\end{prop}
\end{mainbox}

\begin{mainbox}{}
\begin{prop}\label{topos is regular}
A topos is a regular category.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
    A topos is finitely complete. Moreover, by \Cref{colimits are universal in topos} regular epis are pullback stable. By \Cref{topos is finitely cocomplete}, each topos is finitely cocomplete. Then, by \Cref{conditions for factorization regular epi mono}, each morphism admits a (regular epi, mono) factorization. Thus, a topos is a regular category.
\end{proof}
\end{subbox}

Now we aim to transfer the regularity of a topos to its dual, the cotopos.

\begin{mainbox}{}
\begin{prop}\label{mono are equalizers in topos}
    In a topos, every monomorphism is an equalizer.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
    Let $m: S \rightarrowtail A$ be a mono in a topos, let $\chi_S$ be its characteristic morphism and let $t:1 \to \Omega$ be the morphism underlying the subobject classifier. It holds that $m = \mathrm{Eq}(\chi_S,t!^A)$.
    \[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBoAGAXVJADcBDAGwFcYkQBBEAX1PU1z5CKAMwVqdJq3YAdGQHkAtjADm9HnxAZseAkQBM4mgxZtEIYhv46hRckcmnZMxfRwALAE6LgAUQCO3AAUcgDG7lgA+gDKpDgAhAB6HACUPBIwUCrwRKAAZp4Qikj2IDgQSIaO0uZJXLz5hcWIVeVIZNVmZVYgBUVIYmUViKUmNSDFNDj0WIzs07M9fc2lbYgdAEYwYFBIALQio1JdYRExIDSM9FuMAAoCusIgnlgq7jjp3EA
\begin{tikzcd}

\end{tikzcd}
    \]
\end{proof}
\end{subbox}

\begin{mainbox}{}
\begin{prop}\label{mono epi are iso in topos}
    In a topos, a morphism is an isomorphism if and only if it is both a monomorphism and an epimorphism.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
    Clearly, every iso is both a mono and an epi. Conversely, let $f: X \to Y$ be a morphism which is both a mono and an epi. By \Cref{mono are equalizers in topos}, it is also an equalizer. Let $(a,b)$ be pair such that $f=\mathrm{Eq}(a,b)$. It holds $af=bf$. Since $f$ is an epi, then $a=b$. On the other hand, $\mathrm{Eq}(a,b)=\mathrm{Eq}(a,a)=1_Y$, so there exists an iso $g$ such that $fg=1_Y$. Hence $f$ is an iso.
    \[

\end{subbox}

\begin{mainbox}{}
\begin{prop}\label{epi are reg epi in topos}
    In a topos, every epimorphism is a coequalizer.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
    Let $e : A \twoheadrightarrow B$ be an epi in a topos. By \Cref{topos is regular}, a topos is a regular category, thus $e$ admits factorization $e=mr$ where $r$ is a regular epi and $m$ is a mono. It holds that $m$ is also an epi. In fact, if $a$ and $b$ are morphisms such that $am=bm$, then $ae=amr=bmr=be$, and since $e$ is an epi, $a=b$. We proved that $m$ is both a mono and an epi, thus, by \Cref{mono epi are iso in topos}, it is an iso. Hence $e$ is a regular epi, ie a coequalizer.

\end{tikzcd}

\end{proof}
\end{subbox}

\begin{mainbox}{}
\begin{prop}\label{mono are pushot stable}
    In a topos, monomorphisms are pushout stable.
\end{prop}
\end{mainbox}
\noindent The proof is contained in \cite{Johnstone1977} \textit{Corollary 1.28}

\begin{mainbox}{}
\begin{prop}\label{cotopos is regular}
    A cotopos is a regular category.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
Consider a cotopos. It is finitely complete by definition. In the dual category of a cotopos, we aim to prove that each morphism admits a (regular mono, epi) factorization. Let $f: A \to B$ be a morphism and consider its factorization (regular epi, mono) $f=me$ given by \Cref{topos is regular}. $e$ is regular epi, thus, by \Cref{reg epi is an epi}, an epi. $m$ is a mono, thus, by \Cref{mono are equalizers in topos}, it is an equalizer, ie a regular mono. Finally, by \Cref{mono are pushot stable}, monos are pushout stable. Thus, the cotopos is a regular category.
\end{proof}
\end{subbox}

We need to prove that a cotopos is also exact. First, we introduce the notion of subobject. A subobject is a monomorphism into an object. We can define the intersection and union of subobjects.

Given $a:X_1 \rightarrowtail X$ and  $b:X_2 \rightarrowtail X$ subobjects, the intersection is given by the pullback
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0B9ARgAIAdfgGM6aXlwBMIAL6l0mXPkIoJpblVqMWbdjLkgM2PASKrK1es1aIOPPfKNKiZdRa3XbU6RphQA5vBEoABmAE4QALZIAMzUOBBI3HF0WAxsOClpblZsAEb2IGGRSKog8YnJqemZINnaNnQFRVGIZGUJiKUZVTbdabIh4S1t5YixZTW9Nd7SQA
\begin{tikzcd}
X_1 \cap X_2 \arrow[rr, tail] \arrow[d, tail] &  & X_1 \arrow[d, "a", tail] \\
X_2 \arrow[rr, "b", tail]                     &  & X                       
\end{tikzcd}
\]
while the union is given by the monomorphism $m$ underlying the (regular epi, mono) factorization from \Cref{topos is regular} of $[a,b]: X_1 + X_2 \to X$.
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0B9ARgGouATCAC+pdJlz5CKAeSq1GLNuxFiQGbHgJFupbvPrNWiDjwAEAHQsBjJmjOCR8mFADm8IqABmAJwgBbJDIQHAgkXQUjNmQ6UgAjClVvP0DEYNCkWRAACxg6KDYcAHcIXPyEUWSAzOoMxAicOiwGQqaW6kMlE0DqBjo4mAYABQktaRAfLFdsnCdhIA
\begin{tikzcd}
X_1+X_2 \arrow[rr, "{[a,b]}"] \arrow[rd, two heads] &                                     & X \\
                                                    & X_1 \cup X_2 \arrow[ru, "m"', tail] &  
\end{tikzcd}
\]

The union is characterized in term of pushout.
\begin{mainbox}{}
\begin{prop}\label{union is pushout}
    Let $a: X_1 \rightarrowtail X$ and $b: X_2 \rightarrowtail X$ be subobjects of a cotopos $\mathcal{E}$ and their intersection given by
    \[
    % https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0B9ARgAIAdfgGM6aXlwBMIAL6l0mXPkIoJpblVqMWbdjLkgM2PASKrK1es1aIOPPfKNKiZdRa3XbU6RphQA5vBEoABmAE4QALZIAMzUOBBI3HF0WAxsOClpblZsAEb2IGGRSKog8YnJqemZINnaNnQFRVGIZGUJiKUZVTbdWZo5NkJN4S1t5YixZTW9NXUeUDIU0kA
\begin{tikzcd}
X_1 \cap X_2 \arrow[rr, "c", tail] \arrow[d, "d", tail] &  & X_1 \arrow[d, "a", tail] \\
X_2 \arrow[rr, "b", tail]                               &  & X                       
\end{tikzcd}
\]
Then the union $X_1 \cup X_2\rightarrowtail X$ is given by the pushout
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0B9ARgAIAdfgGM6aXlwBMIAL6l0mXPkIoJpblVqMWbLn0FCmYyTLkgM2PASKrK1es1aIOPE-ItKiZdXa2PnU6Q0YKABzeCJQADMAJwgAWyQAZmocCCRuFLosBjYcLJzZKNiExFUQVPTM7Nz81xAY+KQycrTSqpynPOqfBzYhOoaS5orEZPLaztqe7ScoGQppIA
\begin{tikzcd}
X_1 \cap X_2 \arrow[rr, "c", tail] \arrow[d, "d", tail] &  & X_1 \arrow[d, tail] \\
X_2 \arrow[rr, tail]                                    &  & X_1 \cup X_2       
\end{tikzcd}
\]
\end{prop}
\end{mainbox}
\noindent The proof is contained in \cite{Johnstone1977} \textit{Proposition 1.55}.

We are ready to prove the exactness of a cotopos

\begin{mainbox}{}
\begin{prop}\label{cotopos is exact}
    A cotopos is an exact category.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
Consider a cotopos $\mathcal{E}^\mathrm{op}$. By \Cref{cotopos is regular}, it is a regular category. It remains to prove that each equivalence relation is a kernel pair. We prove that each reflexive relation is a kernel pair, then, it follows the result. Consider a reflexive relation on an object $X$ in $\mathcal{E}^\mathrm{op}$, which is the following in $\mathcal{E}$
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBoAGAXVJADcBDAGwFcYkQANEAX1PU1z5CKchWp0mrdgE0e4mFADm8IqABmAJwgBbJKJA4ISMiDgALLGpxIAtACYaDFm0QgNAfWI8+ITTr00hsY05pbWiA4SzuwediA0jPQARjCMAAoCeAQxWIpm1rzqWrqIJkGI+okp6ZlC7Fhg2LDxUVKuCNyU3EA
\begin{tikzcd}
Y \arrow[r, "s" description] & X \arrow[l, "r_1", shift left=2] \arrow[l, "r_2"', shift right=2]
\end{tikzcd}
\]
where $s$ is a common retraction $sr_1=sr_2=1_X$.

Since $(Z,r_1,r_2)$ is a relation in $\mathcal{E}^\mathrm{op}$, then $(r_1,r_2)$ is a jointly monic pair in $\mathcal{E}^\mathrm{op}$. Hence $[r_1,r_2]: X + X \to Y$ is an epi in $\mathcal{E}$. By \Cref{epi are reg epi in topos}, $[r_1,r_2]$ is a regular epi. This means we can assume that $[r_1,r_2]$ is the regular epi underlying the (regular epi, mono) factorization of $[r_1,r_2]$, where the mono is the identity morphism.
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0BqdkAX1PSZc+QigBM5KrUYs2ATT4CQGbHgJEAjKQ1T6zVohALeUmFADm8IqABmAJwgBbJGRA4ISLdP1tkdgPpaAWIUINQAFjB0UGw4AO4QkdEI-LYOzoiu7kgSIEkxhvGJUVApSvZOOdTZiF44dFgMsQ1N1Hqyhhr+xhS8QA
\begin{tikzcd}
X+X \arrow[rr, "{[r_1,r_2]}", two heads] \arrow[rd, two heads] &                           & Y \\
                                                               & Y \arrow[ru, "1_Y", tail] &  
\end{tikzcd}
\]
Thus $Y$ is, up to iso, the union $X \cup X$. Now consider the intersection
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0ACAHW4GM6aTuxABfUuky58hFACZSARiq1GLNgC0xEkBmx4CRBZWr1mrRB22T9MomWWm1Fq6JUwoAc3hFQAMwAnCABbJDIQHAgkBVVzNihrEECQsOpIpABmJzjLPkTk0MQsiKjERWz1SwCAfTl8oMKY9LKKlxq6t1EgA
\begin{tikzcd}
X \cap X \arrow[rr, "d"] \arrow[d, "c"] &  & X \arrow[d, "r_2"] \\
X \arrow[rr, "r_2"]                     &  & Y                 
\end{tikzcd}
\]
Observe that
\[c = 1_X c = sr_1 c = sr_2 d = 1_X d \]
Thus the intersection diagram, which is also a pushout in $\mathcal{E}$ by \Cref{union is pushout}, becomes
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0ACAHW4GM6aTuxABfUuky58hFACZSARiq1GLNgC0xEkBmx4CRBZWr1mrRB22T9MomWWm1Fq6JUwoAc3hFQAMwAnCABbJDIQHAgkBVVzNj5rEECQsOpIpABmJzjLBPF-INDELIioxEVs9UsAgH05ROSimPTyypda+rdRIA
\begin{tikzcd}
X \cap X \arrow[rr, "c"] \arrow[d, "c"] &  & X \arrow[d, "r_2"] \\
X \arrow[rr, "r_2"]                     &  & Y                 
\end{tikzcd}
\]
This diagram is a kernel pair in $\mathcal{E}^\mathrm{op}$ and it is the kernel pair we were looking for.
\end{proof}
\end{subbox}

While a topos fails to be protomodular, a cotopos actually has this structure.

\begin{mainbox}{}
\begin{prop}\label{cotopos is protomodular}
    A cotopos is a protomodular category.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
Consider a cotopos $\mathcal{E}^\mathrm{op}$. By \Cref{exp in topos is monadic} the functor
\[\Omega^{(-)}: \mathcal{E}^\mathrm{op} \to \mathcal{E}\]
is monadic. In particular, one can show that this functor has image in the internal Heyting algebras in $\mathcal{E}$ which is, by \cite{Bourn1996} \textit{Proposition 18}, protomodular. This functor, restricted in the codomain, is still monadic. Then by \Cref{algebras category is proto if the base category is proto}, also $\mathcal{E}^\mathrm{op}$ is protomodular.
\end{proof}
\end{subbox}

It remains to prove that in a cotopos $\mathcal{E}^\mathrm{op}$ the unique morphism 
\[!: 0_{\mathcal{E}^\mathrm{op}} \to 1_{\mathcal{E}^\mathrm{op}}\]
is a regular epi. Then, the associated pullback functor may serve as underlying functor in the definition of an ideally exact category. In order to prove that $!$ is a regular epi in $\mathcal{E}^\mathrm{op}$, we prove, equivalently, that
\[!: 0_\mathcal{E} \to 1_\mathcal{E}\]
is an equalizer.

\begin{mainbox}{}
    \begin{prop}\label{initial object is a subobject}
        In a topos, the initial object $0$ is the subobject of every object.
    \end{prop}
\end{mainbox}

\begin{subbox}{}
    \begin{proof}
        The initial object is a colimit. By \Cref{colimits are universal in topos}, it is pullback stable. In the case of the colimit of the empty diagram, ie the initial object, this means that given any morphism $f: B \to 0$ is an iso. Thus $B$ is the initial object.

        Consider an object $A$. We aim to prove that
        \[!_A: 0 \to A\]
        is a mono. Let $a,b : X \to 0$ such that $!_A a = !_A b$. By the observation above, $X$ is initial, thus $a = b$. Hence $!_A$ is a mono.
    \end{proof}
\end{subbox}  

\begin{mainbox}{}
    \begin{cor}\label{In a topos ! is an equalizer}
        In a topos $\mathcal{E}$, the unique morphism
        \[!: 0_\mathcal{E} \to 1_\mathcal{E}\]
        is an equalizer.
    \end{cor}
\end{mainbox}

\begin{subbox}{}
    \begin{proof}
        By \Cref{initial object is a subobject}, the morphism $!: 0_\mathcal{E} \to 1_\mathcal{E}$ is a mono. Then, by \Cref{mono are equalizers in topos}, it is an equalizer.
    \end{proof}
\end{subbox}

All the results above leads to the following result.

\begin{mainbox}{}
    \begin{prop}\label{cotopos is ideally exact}
        A cotopos is an ideally exact category. Moreover, the underlying functor is the pullback functor.
    \end{prop}
\end{mainbox}

\begin{subbox}{}
    \begin{proof}
        We aim to prove that a cotopos $\mathcal{E}^\mathrm{op}$ satisfies the condition \Cref{Ideally exact categories}\textit{:(1)}. By \Cref{cotopos is exact} it is exact, and by \Cref{cotopos is protomodular} it is protomodular. It has binary coproducts, in fact, the underlying topos is finitely complete. Finally, by \Cref{In a topos ! is an equalizer} the morphism
        \[!: 0_\mathcal{E} \to 1_\mathcal{E}\]
        is an equalizer in $\mathcal{E}$. Thus the morphism 
        \[!: 0_{\mathcal{E}^\mathrm{op}} \to 1_{\mathcal{E}^\mathrm{op}}\]
        is a coequalizer in $\mathcal{E}^\mathrm{op}$, ie a regular epi.
    \end{proof}
\end{subbox}

\begin{rembox}{}
\begin{rem}
    While a cotopos is an ideally exact category, a topos fails to be. A topos is exact and it has binary coproducts, that said, it does not satisfy the remaining conditions. For example one can consider the category $\mathsf{Set}$ which is a topos. It is not protomodular. Moreover, the unique morphism $!: \emptyset \to \{*\}$ is not a regular epi, in fact, it is not even an epi. In order to show that, consider the morphisms
    \begin{align*}
        f: \{*\} & \to \{a,b\} \\
        *       & \mapsto a \\[1.5ex]
        g: \{*\} & \to \{a,b\} \\
        *       & \mapsto b
    \end{align*}
    Then $f! = g!$, since they are both the empty map, but $f \neq g$.
    \[
    % https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAB12YBbNHATzgwcIAL6l0mXPkIoAjOSq1GLNp2AAqTqLESQGbHgJEATIur1mrRB3bA6pAEbaxSmFADm8IqABmAJwhuJDIQHAgkBWUrNgBCXT9A4MQo8KQzEDgACyxfEUQAWijLVRtfBJAAoMjqNMQM7Nz84pVrEA8QagY6RxgGAAUpI1kQfywPLJFRClEgA
\begin{tikzcd}
\emptyset \arrow[r, "!"] & \{*\} \arrow[r, "f", shift left] \arrow[r, "g"', shift right] & {\{a,b\}}
\end{tikzcd}
    \]
\end{rem}
\end{rembox}


\section{The dual of sets}

In the previous section we have seen that a cotopos is an ideally exact category by \Cref{cotopos is ideally exact}. An example of cotopos is the dual of the category of sets $\mathsf{Set}^\mathrm{op}$. In this section we aim to analyze this case.

As in the case of cotopos, we refer to \Cref{Ideally exact categories}\textit{(1)} for the definition of ideally exact categories.

First we want to show directly that
\[!: \emptyset \to \{*\}\]
is an equalizer in the underlying topos. So that, it is a regular epi in the cotopos.

In $\mathsf{Set}$ the equalizer of a pair of parallel arrows $f,g: A \to B$ is
\[  \mathrm{Eq}(f,g) = \{ x \in A \mid f(x) = g(x) \} \]
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAB12BbOnACwBOXYAFEAjgF8AFADNSAcwCUICaXSZc+QigCM5KrUYs2AQRVqQGbHgJEATPur1mrRCABCKgzCjz4RUBkBCC4kPRAcCCQHEDg+LBkcJABacOdjNxlzQODQxHDI6Oo4hKT8pyNXEHkQagY6ACMYBgAFDRttEAEseT4k1RyQpDIIqPyJCgkgA
\begin{tikzcd}
{\mathrm{Eq}(f,g)} \arrow[r, hook, "i"] & A \arrow[r, "f", shift left] \arrow[r, "g"', shift right] & B
\end{tikzcd}
\]
Consider in particular the morphisms
    \begin{align*}
        f: \{*\} & \to \{a,b\} \\
        *       & \mapsto a \\[1.5ex]
        g: \{*\} & \to \{a,b\} \\
        *       & \mapsto b
    \end{align*}
    Clearly $f \neq g$, thus
    \[\mathrm{Eq}(f,g) = \emptyset\]
    Since $!$ is the unique morphism from the initial object to any object, it corresponds in this case with the inclusion $i$. Thus $!$ is the equalizer of $f$ and $g$.
    \[
    % https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAB12YBbNHATzgwcIAL6l0mXPkIoAjOSq1GLNp2AAqTqLESQGbHgJEATIur1mrRB3bA6pAEbaxSmFADm8IqABmAJwhuJDIQHAgkBWUrNgBCXT9A4MQo8KQzEDgACyxfEUQAWijLVRtfBJAAoMjqNMQM7Nz84pVrEA8QagY6RxgGAAUpI1kQfywPLJFRClEgA
\begin{tikzcd}
\emptyset \arrow[r, "!"] & \{*\} \arrow[r, "f", shift left] \arrow[r, "g"', shift right] & {\{a,b\}}
\end{tikzcd}
    \]

    In order to present $\mathsf{Set}^\mathrm{op}$ as an ideally exact category and study the corresponding ideals, we aim to understand which is the underlying semiabelian category. Keep in mind that the morphism $!$, in $\mathsf{Set}^\mathrm{op}$, is $!: \{*\} \to \emptyset$. As shown in \Cref{Ideally exact categories}$\mathit{:(1\implies 3)}$ such semiabelian category is
    \[ \mathcal{X} = (\mathsf{Set}^\mathrm{op} \downarrow \{*\}) \]
    where the objects are the object of the coslice category from the singleton to sets
    \[ \mathrm{Ob}(\mathcal{X}) = \mathrm{Ob}(\mathsf{Set}^\mathrm{op} \downarrow \{*\}) =  \mathrm{Ob}(\{*\}\downarrow \mathsf{Set}) \]
    or, equivalently, of the category of pointed sets
    \[ \mathrm{Ob}(\mathcal{X}) = \{(A, a) \ | \ A \in \mathsf{Set}, \ a \in A \} \]
    and the morphism are the morphism of $\mathsf{Set}^\mathrm{op}$ preserving the chosen element. Summing up
    \begin{align*}
        \mathrm{Ob}(\mathcal{X}) &= \{(A, a) \ | \ A \in \mathsf{Set}, \ a \in A \} \\[1.5ex]
        \mathrm{Hom}_{\mathcal{X}}((A,a),(B,b)) &= \{ f : B \to A \ | \ f(b)=a \} \subseteq \mathrm{Hom}_{\mathsf{Set}^\mathrm{op}}
    \end{align*}
    Thus, finally
    \[\mathcal{X} \cong (\mathsf{Set}^\bullet)^\mathrm{op} \]
    where $\mathsf{Set}^\bullet$ is the category of pointed sets, ie the category of sets with a chosen element.
    
    This category is, actually, pointed in the categorical meaning, in fact, the singleton $(\{*\},*) \cong \{*\}$ is both the initial and terminal object. Given an object $(A,a) \in \mathcal{X}$ we have the following unique morphism from the singleton and to the singleton, represented in $\mathsf{Set}$.
    \begin{align*}
    \{*\} &\to (A,a) &\quad (A,a) &\to \{*\} \\[1ex]
    *     &\mapsto a &\quad (x \in A)     &\mapsto *
    \end{align*}
    Moreover, the zero morphism in $\mathcal{X}$ turns out to be the constant morphism to the chosen element
    \begin{align*}
    0_{(A,a), (B,b)}: (B,b) &\to (A,a) \\[1ex]
    (x \in B) &\mapsto a
    \end{align*}
    
    To understand the action of the underlying functor 
    \[  U : \mathsf{Set}^\mathrm{op} \to \mathcal{X} \cong (\mathsf{Set}^\bullet)^\mathrm{op} \]
    of the ideally exact category, we need to analyze the pullback functor of the morphism $!$. It is defined through the pullback $!$ along the morphism given with an object in $\mathsf{Set}^\mathrm{op}$. Such pullback, described in the second chapter, corresponds, given an object $A$ in $\mathsf{Set}^\mathrm{op}$, to the pushout in $\mathsf{Set}$
    \[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBoBGAXVJADcBDAGwFcYkQAdD4AKi4F8Q-Uuky58hFACYK1Ok1bsuMALZocATzgwcQkSAzY8BIjOJyGLNohABBPaKMSiZczUuKbACi68BAAi44AEcAY2Y0f1sASgB6QI48RlghORgoAHN4IlAAMwAnCBUkchocCCQyeSt2AEIQGkZ6ACMYRgAFMWNJEHysDIALXWE8wuLEUpBypCkRkAKiyrKKxABmdwVrTg58HHoAfXIHebGZ5aR16s9t3YOpBpAm1o6u5xs+weHKfiA

\{*\} \arrow[u, "\iota_1"] &  & \emptyset \arrow[ll, "!"'] \arrow[u]
\end{tikzcd}
\]
where $\sim$ is the smallest equivalence relation identifying the image of any element through the empty function $\emptyset \to A$ and $!$. The injection $\iota_1$ and $\iota_2$ are the canonical inclusions in the disjoint union, followed by the quotient map. In this case the definition of the $\sim$ is based in the empty set, thus $\sim$ is the empty relation and the pushout is just the disjoint union of the singleton and $A$. Thus we obtain
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBoBGAXVJADcBDAGwFcYkQAdD4AKi4F8Q-Uuky58hFACYK1Ok1bsuMALZocATzgwcQkSAzY8BIjOJyGLNohABBPaKMSiZczUuKbXXgIAEXOABHAGNmNF97fjkYKABzeCJQADMAJwgVJHIaHAgkMnkrdgBCEBpGegAjGEYABTFjSRAUrFiAC11hZLSMxCyQHKQpTpBU9Lzs3MQAZncFa04OfBx6AH1yBxHuwYmkGYLPBaXVqVKQcqra+ucbZraOyn4gA
\begin{tikzcd}
\{*\} \sqcup A             &  & A \arrow[ll, "\iota_2"']             \\
\{*\} \arrow[u, "\iota_1"] &  & \emptyset \arrow[ll, "!"'] \arrow[u]
\end{tikzcd}
\]
Then, the pullback functor associates to an object $A$ in $\mathsf{Set}^\mathrm{op}$ the object $(\{*\} \sqcup A, \iota_1)$ in $\mathcal{X}$. By the identification above, we can write this as
\begin{align*} 
    U : \mathsf{Set}^\mathrm{op} &\to (\mathsf{Set}^\bullet)^\mathrm{op}  \\[1ex]
    A &\mapsto ( \{*\} \sqcup A, *)
\end{align*}
in fact $(\{*\} \sqcup A, \iota_1)$ corresponds to the pointed set $(\{*\} \sqcup A,*)$.

The functor $U$ is monadic by the Beck monadicity theorem. In fact, consider $(f,g)$ a pair of parallel morphism in $\mathsf{Set}^\mathrm{op}$. Then, consider their coequalizer in $\mathsf{Set}^\mathrm{op}$, which is actually the equalizer in $\mathsf{Set}$. 
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBoAGAXVJADcBDAGwFcYkQBBEAX1PU1z5CKAEwVqdJq3YAhHnxAZseAkXLiaDFm0QgAOnuAAPAAQGsYExzN6TAHxsmAZgAojASgC8Aczfub3DwSMFDe8ESgTgBOEAC2SOogOBBIZCBwABZYTjhIALRpWtK6TvKRMfGIicmpNJnZuYiFUjog3iA0jPQARjCMAAoCKsIgUVjeGbm85XFIYkkpVdyU3EA

                                                                                                                             \\

\end{tikzcd}
\]
Thus the hypothesis of the Beck monadicity theorem is satisfied.

Now, we aim to understand what are the ideals given by \Cref{quotients in ideally exact} in the case of $\mathsf{Set}^\mathrm{op}$.

By the mentioned theorem, given an object $A$ in $\mathsf{Set}^\mathrm{op}$, we have that
\[\mathrm{Quot}(A) \cong \mathrm{NSub}(U(A))\]
\[
\left\{
\begin{array}{c}
\text{Classes of iso of} \\
\text{regular epis } \mathsf{Set}^\mathrm{op} \\
\text{with domain} \ A

\right\} \ \ \ \ \ \ \ \  \longleftrightarrow \ \ \ \ \ \ \ \

\vspace{1em}
\noindent $\mathrm{Quot}(A)$ are the regular epi in $\mathsf{Set}^\mathrm{op}$, which are the equalizers in $\mathsf{Set}$. Equalizers in $\mathsf{Set}$ are all the monomorphisms, ie all the injective functions. In fact, every equalizer is a mono. Conversely, given a mono $m: B \rightarrowtail A$, we can consider the pair of morphism
    \begin{align*}
        f: A & \to \{0,1\} \\[1ex]
           x    & \mapsto 1 \\[1.5ex]
        g: A & \to \{0,1\} \\[1ex]
           x    & \mapsto
\begin{cases}
  1 & \text{if } x \in \mathrm{Im}(m) \\
  0 & \text{otherwise}
\end{cases}
\end{align*}
Now, their equalizer is
\[\mathrm{Eq}(f,g) = \{x \in A \ | \ f(x) = g(x) \} = \{x \in A \ | \ g(x) = 1 \} = \mathrm{Im}(m)\]
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBoAGAXVJADcBDAGwFcYkQBBEAX1PU1z5CKAEwVqdJq3YAdGcHKlic7jz4gM2PASKKqNBizaIQcgLb0cACwBOZ4AFEAjtwAUAM1IBzAJQ8JMFBe8ESg7jYQZkiKIDgQSGQgcFZY7jhIALSJhtIm7mphEVGIMXEJNMmp6YjZUsYgXiA0jPQARjCMAAoC2sIgNlheVum8hZFIYrHxJdyU3EA
\begin{tikzcd}
{\mathrm{Im}(m)} \arrow[r] & A \arrow[r, "f", shift left] \arrow[r, "g"', shift right] & {\{0,1\}}
\end{tikzcd}
\]
Then, since $m$ induces an iso between $\mathrm{Im}(m)$ and $B$, $m$ is the equalizer of $(f,g)$.
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAEEQBfU9TXfIRQBGclVqMWbAELdeIDNjwEiAJjHV6zVohAAdPcDLCDXbuJhQA5vCKgAZgCcIAWyRkQOCElETtbNx4HZzdEXy8kdRA4AAssexwkAFpfLSldezlg1x9qCMQo2PjEsM1JHRArEGoGOgAjGAYABX5lIRBHLCsYxK4KLiA
\begin{tikzcd}
B \arrow[r, "m"] & A \arrow[r, "f", shift left] \arrow[r, "g"', shift right] & {\{0,1\}}
\end{tikzcd}
\]
Thus $\mathrm{Quot}(A)$ consists in $\mathsf{Set}$ of all the injective functions with codomain $A$.

On the other hand, $\mathrm{NSub}(U(A))$ are the kernels of some morphism in $(\mathsf{Set}^\bullet)^\mathrm{op}$ with domain $U(A)$. Recall that $U(A) = (\{*\} \sqcup A, *)$. Thus $\mathrm{NSub}(U(A))$ are the cokernels of some morphism in $\mathsf{Set}^\bullet$ with codomain $(\{*\} \sqcup A, *)$.

We can conclude that the class of iso of injective functions with codomain $A$ is in bijection with the class of iso of cokernels with domain $(\{*\} \sqcup A, *)$.
\[
\left\{
\begin{array}{c}
\text{Classes of iso of} \\
\text{injective functions in } \mathsf{Set} \\
\text{with codomain} \ A
\end{array}
\right\} \ \  \longleftrightarrow \ \ 
\left\{
\begin{array}{c}
\text{Classes of iso of} \\
\text{cokernels in } \mathsf{Set^\bullet} \\
\text{with domain} \ (\{*\} \sqcup A, *)
\end{array}
\right\}
\]
