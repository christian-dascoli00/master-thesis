\chapter{Cotopos as an ideally exact category}

\section{The general case}

In this section, we aim to show that any cotopos $\mathcal{E}^{\mathrm{op}}$, the dual of a topos $\mathcal{E}$, is an ideally exact category. Then, since the category of $\mathbf{Set}$ is a topos, we will analyze the case of $\mathbf{Set}^{\mathrm{op}}$. The following is based on \cite{Bourn2004} and \cite{Borceux1994II}, \cite{Borceux1994III}.

First, we recall the definition of a topos.

\begin{mainbox}{}
\begin{defi}
Let $\mathcal{E}$ be a finitely complete category. A \textbf{subobject classifier} of $\mathcal{E}$ is a pair $(\Omega, t)$, where $\Omega$ is an object of $\mathcal{E}$ and $t: 1 \rightarrowtail \Omega$ is a monomorphism, such that for any object $A$ of $\mathcal{E}$ and any monomorphism $m: S \rightarrowtail A$, there exists a unique morphism $\chi_S: A \rightarrow \Omega$ making the diagram
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAGUQBfU9TXfIRQBGclVqMWbADrSAtnRwALAEYAzYMK7deIDNjwEiZYePrNWiEAEEdfA4KKjT1c1KuyA8nJgBzOtziMFC+8ESgagBOEHJIZCA4EEhaulExcdSJSABMmXRYDGw4+YWukpYgsdQMdCowDAAK-IZCIJFYvko4diBpsYi5CUmIAMxlFjLSAMZKWAD6nDwR0f2iQ0hjEhNW3VwUXEA
\begin{tikzcd}
S \arrow[r] \arrow[d, "m"', tail] & \mathbf{1} \arrow[d, "t"] \\
A \arrow[r, "\chi_S"]             & \Omega                   
\end{tikzcd}
\]
a pullback.
\end{defi}
\end{mainbox}

\newpage

\begin{mainbox}{}
\begin{defi}
A category $\mathcal{E}$ is said to be an \textbf{elementary topos}, in the following just a \textbf{topos}, if
\begin{itemize}
    \item it is finitely complete;
    \item it is cartesian closed;
    \item it has a subobject classifier.
\end{itemize}
\end{defi}
\end{mainbox}

A topos turns out to have a rich structure. First, we need some properties. Recall that the exponential is given by the cartesian closedness: for all objects $A$, $B$, and $C$ in $\mathcal{E}$.
\[
\mathrm{Hom}_{\mathcal{E}}(A, \Omega^B) \cong \mathrm{Hom}_{\mathcal{E}}(A \times B, \Omega) \cong \mathrm{Hom}_{\mathcal{E}}(B, \Omega^A)
\]

\begin{mainbox}{}
\begin{prop}\label{exp in topos is monadic}
Given a topos, the exponential functor 
\[\Omega^{(-)}: \mathcal{E}^{\mathrm{op}} \to \mathcal{E} \]
is monadic.
\end{prop}
\end{mainbox}
\noindent The proof is contained in \cite{Borceux1994III} \textit{Thm 5.7.3} and it is based on the Beck monadicity theorem.

\begin{mainbox}{}
\begin{prop}\label{topos is finitely cocomplete}
A topos is finitely cocomplete.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
Consider a topos $\mathcal{E}$. Since the functor $\Omega^{(-)}: \mathcal{E}^{\mathrm{op}} \to \mathcal{E}$, by \Cref{exp in topos is monadic}, is monadic, and $\mathcal{E}$ is finitely complete, it follows that $\mathcal{E}^{\mathrm{op}}$ is finitely complete. Thus, $\mathcal{E}$ is finitely cocomplete.
\end{proof}
\end{subbox}

\begin{mainbox}{}
\begin{prop}\label{conditions for factorization regular epi mono}
Let $\mathcal{E}$ be a category with the following conditions:
\begin{itemize}
\item it has pullbacks;
\item it has coequalizers of kernel pairs;
\item it has pullback stable regular epimorphisms.
\end{itemize}
Then each morphism $f: A \to B$ in $\mathcal{E}$ admits a (regular epi, mono) factorization.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
Let $f: A \to B$ be a morphism. Consider the kernel pair $(r_1,r_2)$ of $f$, and the coequalizer $e$ of the kernel pair.
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBoAGAXVJADcBDAGwFcYkQBBEAX1PU1z5CKAMwVqdJq3YAhHnxAZseAkQBMpYhIYs2iEAEl5-ZUKLlxNHdP0cAOnbwBbeAH0ZXbhJhQA5vCJQADMAJwgnJAsQHAgkMklddiDjEFDwyJoYpA0EmxA2GkZ6ACMYRgAFARVhEBCsXwALHBS0iMQxaNjEKLgGrCDmxABaYl5gsLaOrO6aXv7B0cpuIA
\begin{tikzcd}
A\times_BA \arrow[r, shift left] \arrow[r, shift right] & A \arrow[rr, "f"] \arrow[rd, "e"'] &   & B \\
                                                        &                                    & I &  
\end{tikzcd}
\]
Then, since $fr_1=fr_2$, by the universal property of the coequalizer, there exists a unique morphism $m$ such that $me = f$. We already know that $e$ is a regular epi, since it is a coequalizer.

It remains to prove that $m$ is a mono. Equivalently, we show that, letting $p_1, p_2: I \times_B I \to I$ be the projections of the kernel pair, we have that $p_1 = p_2$. First, we consider the morphism $e\times_Be : A\times_BA \to I\times_BI$.
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBoAGAXVJADcBDAGwFcYkQBBEAX1PU1z5CKAMwVqdJq3YAhHnxAZseAkQBMpYhIYs2iEAEl5-ZUKLlxNHdP0cAOnbwBbeAH0ZXXicGqUZNdpSeoYOzm4yRtwSMFAA5vBEoABmAE4QTkgWIDgQSGSSuuxJxiCp6Zk0OUgaBTYgbDSM9ABGMIwACgIqwiApWLEAFjglZRmIYtm5iFlwA1hJw4gAtMRepWljE1XTNLPzi6sKo9WVU-mMWGDBcBAXUCBWQewZjS1tnaa+vf1DIxtIABZTicQHsFkgVmtjoggZMQWCDlD-uNgTDHoV9DBQlgXHB3A0QE1Wh0umZ9H1BsMotwgA
\begin{tikzcd}
A\times_BA \arrow[r, shift left] \arrow[r, shift right] \arrow[rdd, "e\times_Be"'] & A \arrow[rr, "f"] \arrow[rd, "e"']                        &                    & B \\
                                                                                   &                                                           & I \arrow[ru, "m"'] &   \\
                                                                                   & I\times_BI \arrow[ru, shift left] \arrow[ru, shift right] &                    &  
\end{tikzcd}
\]
Since $e$ is a coequalizer, it is a regular epi. Thus $e\times_Be$ is an epi. In fact, all the square in the following diagram are pullbacks, and $e\times_Be$ is the composition of the two regular epis, thus it is an epi.
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAEEAdTvAW3gD6AIXYgAvqXSZc+QigBM5KrUYs2ASW59BI8ZJAZseAkQDMS6vWatEHPVKOyzpecqtrb6+wenG5yRVM3VRsQLwkHGRMUMiDLELZRCJ9HaORzOJVrNiFvQyj-Mld47NsuHix+OGFw-Xy-IkVirI8wrUqdcOUYKABzeCJQADMAJwheJDIQHAgkAEZqAAsYOig2HAB3CGXVhGTR8cnqGaQAdiWVtdtN7cu9-QOJxHPp2cQADgvV9a2dqHvhmMngtXkhPiA-j9brtvI8jqCPtQGFgwKEoHQ4Ms1iVWjB2lVhKx9kD5sc3vJiYdEIoEeYWqFWF8rtNfndYSTEbScaE0AI5uyqeCTogACzcti8+QCp5ihEANnFtgmlKedOFCvpbGVDw5L2FAFYVUh9WSkLL3AyQEyoX8ASA4YgpsKaQAjGBgK6mKYWtgjPnS+EG6huj1IAC0IoAnIqQH6pUi6G6GAAFXxOWwjLC9RY4cQUMRAA
\begin{tikzcd}
A\times_BA \arrow[rr, two heads] \arrow[dd, two heads] \arrow[rrdd, "e\times_Be", dashed] \arrow[rrr, "r_1", bend left] \arrow[ddd, "r_2"', bend right=49] &  & I\times_BA \arrow[dd, two heads] \arrow[r]   & A \arrow[dd, "e", two heads] \\
                                                                                                                                                           &  &                                              &                              \\
A\times_BI \arrow[rr, two heads] \arrow[d]                                                                                                                 &  & I\times_BI \arrow[r, "p_1"] \arrow[d, "p_2"] & I \arrow[d, "m"]             \\
A \arrow[rr, "e", two heads]                                                                                                                               &  & I \arrow[r, "m"]                             & B                           
\end{tikzcd}
\]
Finally
\[p_1 (e\times_Be) = e r_1 = e r_2 = p_2 (e\times_Be)\]
\[\implies p_1 = p_2\]
\end{proof}
\end{subbox}

\begin{mainbox}{}
\begin{prop}\label{colimits are universal in topos}
    In a topos, all finite colimits are pullback stable.
\end{prop}
\end{mainbox}
\noindent The proof is contained in \cite{Borceux1994II} \textit{Proposition 5.9.1}.

\begin{mainbox}{}
\begin{prop}
A topos is a regular category.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}\label{topos is regular}
    A topos is finitely complete. Moreover, by \Cref{colimits are universal in topos} regular epis are pullback stable. By \Cref{topos is finitely cocomplete}, each topos is finitely cocomplete. Then, by \Cref{conditions for factorization regular epi mono}, each morphism admits a (regular epi, mono) factorization. Thus, a topos is a regular category.
\end{proof}
\end{subbox}

Now we aim to transfer the regularity of a topos to its dual, the cotopos.

\begin{mainbox}{}
\begin{prop}\label{mono are equalizers in topos}
    In a topos, every monomorphism is an equalizer.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
    Let $m: A \rightarrowtail B$ be a monomorphism in a topos, let $\chi_A$ be its characteristic morphism and let $t:1 \to \Omega$ be the morphism underlying the subobject classifier. It holds that $m = \mathrm{Eq}(\chi_S,t!^A)$.
    \[
    % https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBoAGAXVJADcBDAGwFcYkQBBEAX1PU1z5CKAMwVqdJq3YAdGQHkAtjADm9HnxAZseAkQBM4mgxZtEIYhv46hRckcmnZMxfRwALAE6LgAUQCO3AAUcgDG7lgA+gDKpDgAhAB6HACUPBIwUCrwRKAAZp4Qikj2IDgQSIaO0uZJXLz5hcWIVeVIZNVmZVYgBUVIYmUViKUmNSDFDb1NJTRtiB0ARjBgUEgAtCKjUl1hETEgNIz0y4wACgK6wiCeWCruOOncQA
\begin{tikzcd}
{\mathrm{Eq}(\chi_S,t!^A)} \arrow[r, "m"] & A \arrow[r, "!^A"] \arrow[rr, "\chi_S"', bend right] & 1 \arrow[r, "t"] & \Omega
\end{tikzcd}
    \]
\end{proof}
\end{subbox}

\begin{mainbox}{}
\begin{prop}\label{epi are reg epi in topos}
    In a topos, every epimorphism is a coequalizer.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
    Every epi is the coequalizer of its kernel pair.
\end{proof}
\end{subbox}

\begin{mainbox}{}
\begin{prop}\label{mono are pushot stable}
    In a topos, monomorphisms are pushout stable.
\end{prop}
\end{mainbox}

\begin{mainbox}{}
\begin{prop}\label{cotopos is regular}
    A cotopos is a regular category.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
Consider a cotopos. It is finitely complete by definition. In the dual category, we aim to prove that each morphism admits a (regular mono, epi) factorization. Let $f: A \to B$ be a morphism and consider its factorization (regular epi, mono) $f=me$ given by \Cref{topos is regular}. $e$ is regular epi, thus, by \Cref{reg epi is an epi}, an epi. $m$ is a mono, thus, by \Cref{mono are equalizers in topos}, it is an equalizer, ie a regular mono. Finally, by \Cref{mono are pushot stable}, monos are pushout stable. Thus, the cotopos is a regular category.
\end{proof}
\end{subbox}

We need to prove that a cotopos is also exact. First, we introduce the notion of subobject. A subobject is a monomorphism into an object. We can define the intersection and union of subobjects.

Given $a:X_1 \rightarrowtail X$ and  $b:X_2 \rightarrowtail X$ subobjects, the intersection is given by the pullback
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0B9ARgAIAdfgGM6aXlwBMIAL6l0mXPkIoJpblVqMWbdjLkgM2PASKrK1es1aIOPPfKNKiZdRa3XbU6RphQA5vBEoABmAE4QALZIAMzUOBBI3HF0WAxsOClpblZsAEb2IGGRSKog8YnJqemZINnaNnQFRVGIZGUJiKUZVTbdabIh4S1t5YixZTW9Nd7SQA
\begin{tikzcd}
X_1 \cap X_2 \arrow[rr, tail] \arrow[d, tail] &  & X_1 \arrow[d, "a", tail] \\
X_2 \arrow[rr, "b", tail]                     &  & X                       
\end{tikzcd}
\]
while the union is given by the monomorphism $m$ underlying the (regular epi, mono) factorization from \Cref{topos is regular} of $[a,b]: X_1 + X_2 \to X$.
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0B9ARgGouATCAC+pdJlz5CKAeSq1GLNuxFiQGbHgJFupbvPrNWiDjwAEAHQsBjJmjOCR8mFADm8IqABmAJwgBbJDIQHAgkXQUjNmQ6UgAjClVvP0DEYNCkWRAACxg6KDYcAHcIXPyEUWSAzOoMxAicOiwGQqaW6kMlE0DqBjo4mAYABQktaRAfLFdsnCdhIA

\end{tikzcd}
\]

\begin{prop}\label{union is pushout}
    Let $a: X_1 \rightarrowtail X$ and $b: X_2 \rightarrowtail X$ be subobjects of a cotopos $\mathcal{E}$ and their intersection given by
    \[

We are ready to prove the exactness of a cotopos

\begin{mainbox}{}
\begin{prop}\label{cotopos is exact}
    A cotopos is an exact category.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
Consider a cotopos $\mathcal{E}^\mathrm{op}$. By \Cref{cotopos is regular}, it is a regular category. It remains to prove that each equivalence relation is a kernel pair. We prove that each reflexive relation is a kernel pair, then, it follows the result. Consider a reflexive relation on an object $X$ in $\mathcal{E}^\mathrm{op}$, which is the following in $\mathcal{E}$
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBoAGAXVJADcBDAGwFcYkQANEAX1PU1z5CKchWp0mrdgE0e4mFADm8IqABmAJwgBbJKJA4ISMiDgALLGpxIAtACYaDFm0QgNAfWI8+ITTr00hsY05pbWiA4SzuwediA0jPQARjCMAAoCeAQxWIpm1rzqWrqIJkGI+okp6ZlC7Fhg2LDxUVKuCNyU3EA
\begin{tikzcd}
Y \arrow[r, "s" description] & X \arrow[l, "r_1", shift left=2] \arrow[l, "r_2"', shift right=2]
\end{tikzcd}
\]
where $s$ is a common retraction $sr_1=sr_2=1_X$.

Since $(Z,r_1,r_2)$ is relation in $\mathcal{E}^\mathrm{op}$, then $(r_1,r_2)$ is a jointly monic pair in $\mathcal{E}^\mathrm{op}$. Hence $[r_1,r_2]: X + X \to Y$ is an epi in $\mathcal{E}$. By \Cref{epi are reg epi in topos}, $[r_1,r_2]$ is a regular epi. This means we can assume that $[r_1,r_2]$ is the regular epi underlying the (regular epi, mono) factorization of $[r_1,r_2]$, where the mono is the identity morphism.
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0BqdkAX1PSZc+QigBM5KrUYs2ATT4CQGbHgJEAjKQ1T6zVohALeUmFADm8IqABmAJwgBbJGRA4ISLdP1tkdgPpaAWIUINQAFjB0UGw4AO4QkdEI-LYOzoiu7kgSIEkxhvGJUVApSvZOOdTZiF44dFgMsQ1N1Hqyhhr+xhS8QA
\begin{tikzcd}
X+X \arrow[rr, "{[r_1,r_2]}", two heads] \arrow[rd, two heads] &                           & Y \\
                                                               & Y \arrow[ru, "1_Y", tail] &  
\end{tikzcd}
\]
Thus $Y$ is, up to iso, the union $X \cup X$. Now consider the intersection
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0ACAHW4GM6aTuxABfUuky58hFACZSARiq1GLNgC0xEkBmx4CRBZWr1mrRB22T9MomWWm1Fq6JUwoAc3hFQAMwAnCABbJDIQHAgkBVVzNihrEECQsOpIpABmJzjLPkTk0MQsiKjERWz1SwCAfTl8oMKY9LKKlxq6t1EgA
\begin{tikzcd}
X \cap X \arrow[rr, "d"] \arrow[d, "c"] &  & X \arrow[d, "r_2"] \\
X \arrow[rr, "r_2"]                     &  & Y                 
\end{tikzcd}
\]

Observe that
\[c = 1_X c = sr_1 c = sr_2 d = 1_X d \]
Thus the intersection diagram, which is also a pushout in $\mathcal{E}$ by \Cref{union is pushout}, becomes
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0ACAHW4GM6aTuxABfUuky58hFACZSARiq1GLNgC0xEkBmx4CRBZWr1mrRB22T9MomWWm1Fq6JUwoAc3hFQAMwAnCABbJDIQHAgkBVVzNj5rEECQsOpIpABmJzjLBPF-INDELIioxEVs9UsAgH05ROSimPTyypda+rdRIA
\begin{tikzcd}
X \cap X \arrow[rr, "c"] \arrow[d, "c"] &  & X \arrow[d, "r_2"] \\
X \arrow[rr, "r_2"]                     &  & Y                 
\end{tikzcd}
\]
This diagram is a kernel pair in $\mathcal{E}^\mathrm{op}$ and it is the kernel pair we were looking for.
\end{proof}
\end{subbox}

While a topos fails to be protomodular, a cotopos actually has this structure.

\begin{mainbox}{}
\begin{prop}\label{cotopos is protomodular}
    A cotopos is a protomodular category.
\end{prop}
\end{mainbox}

\begin{subbox}{}
\begin{proof}
Consider a cotopos $\mathcal{E}^\mathrm{op}$. By \Cref{exp in topos is monadic} the functor
\[\Omega^{(-)}: \mathcal{E}^\mathrm{op} \to \mathcal{E}\]
is monadic. In particular, one can show that this functor has image in the internal Heyting algebras in $\mathcal{E}$ which is protomodular. This functor, restricted in the codomain, is still monadic. Then by \Cref{algebras category is proto if the base category is proto}, also $\mathcal{E}^\mathrm{op}$ is protomodular.
\end{proof}
\end{subbox}

It remains to prove that in a cotopos $\mathcal{E}^\mathrm{op}$ the unique morphism 
\[!: 0_{\mathcal{E}^\mathrm{op}} \to 1_{\mathcal{E}^\mathrm{op}}\]
is a regular epi. Then, the associated pullback functor may serve as underlying functor in the definition of an ideally exact category. In order to prove that $!$ is a regular epi in $\mathcal{E}^\mathrm{op}$, we prove, equivalently, that
\[!: 0_\mathcal{E} \to 1_\mathcal{E}\]
is an equalizer.

\begin{mainbox}{}
    \begin{prop}\label{In a cotopos ! is an equalizer}
        In a cotopos ${\mathcal{E}^\mathrm{op}}$ the unique morphism
        \[!: 0_\mathcal{E} \to 1_\mathcal{E}\]
        is an equalizer.
    \end{prop}
\end{mainbox}
