\begin{mainbox}{}
    \begin{defi}
        Let $\gamma$ be a natural transformation between two functors $F,G : \mathcal{C} \to \mathcal{D}$, we say that $\gamma$ is \textbf{cartesian} if for every morphism $f : A \to B$ in $\mathcal{C}$, the following diagram is a pullback:
    \[
    % https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRADEAKAQQEoQAvqXSZc+QigBM5KrUYs2AcR78hI7HgJFpARln1mrRCGUAhVcJAYN4omT3UDC413ODZMKAHN4RUADMAJwgAWyQyEBwIJB1HeSMQAB1ErzoQkLoAfWBuAUFLINDw6iikAGY4wzYufwsA4LDECsjoxGk5KuNk1PSs03z6osRYlqR2pwTlWvcBIA
    \begin{tikzcd}
        F(A) \arrow[rr, "\gamma_{A}"] \arrow[d, "F(f)"] &  & G(A) \arrow[d, "G(f)"] \\
        F(B) \arrow[rr, "\gamma_B"]                     &  & G(B)                  
        \end{tikzcd}
    \]
    ie the naturality diagrams are pullbacks.
    \end{defi}
\end{mainbox}


\begin{mainbox}{}
    \begin{defi}
        Let $\mathcal{C}$ be a category and $f : A \to B$. We say that $f$ is a \textbf{strong epimorphism} if it for each commutative diagram of solid arrows
    \[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAEEQBfU9TXfIRQBGclVqMWbAELdeIDNjwEiZYePrNWiEAGE5fJYKKj11TVJ0ARbuJhQA5vCKgAZgCcIAWyRkQOCCRRCS02VwMQD28kACZqAKQAZni6LAY2HFT080ltEB8eN08fRD8ExDiQyxAmEGoGOgAjGAYABX5lIRB3LAcACxwIqJLg8uSqvJoh4qD4wIr6rDA8qDo4Pvs6ibYAHR2YAA8sOBw4AAIAQjOoLYbmto7jHR7+wa4KLiA
\begin{tikzcd}
    A \arrow[r, "f"] \arrow[d, "u"'] & B \arrow[d, "v"] \arrow[ld, "\exists ! d"', dashed] \\
    C \arrow[r, "m", tail]           & D                                                  
    \end{tikzcd}
    \]
    where $m$ is a mono, there exists a unique morphism $d$ such that $df=u$ and $md=v$.
    \end{defi}
\end{mainbox}

\begin{mainbox}{}
    \begin{prop}\label{Eqivalence strong epi with iso}
    Let $\mathcal{C}$ be a category with pullbacks and $f : A \to B$. The following are equivalent:
    \begin{enumerate}
        \item $f$ is a strong epimorphism;
        \item Given a factorization $f=mg$, if $m$ is a mono, then it is an iso.
    \end{enumerate}
    \end{prop}
\end{mainbox}

\begin{subbox}{}
    \begin{proof}
        $(1 \implies 2)$ Let $f=mg$ be a factorization with $m$ mono and consider the square
        \[
        % https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAEEQBfU9TXfIRQBGclVqMWbAELdeIDNjwEio4ePrNWiELJ58lgomXXVNUnQGFu4mFADm8IqABmAJwgBbJGRA4ISKISWmwucq4e3oi+-kgAzGaS2iD2INQMdABGMAwACvzKQiBuWPYAFjjhIO5e8dSxiABMiSE63ulZOfmGKjol5ZX1dFgMbDjDo-rVkYH1AU3pWGDJcBAMWFBpIGUwdJuIYEwMDEMjbJDLW3BlWC6V0S0WIMIA+nryNVFBDQnBT5sdbJ5ApGPqlCo2LhAA
\begin{tikzcd}
    A \arrow[r, "f"] \arrow[d, "g"'] & B \arrow[d, "1_B"] \arrow[ld, "d"'] \\
    C \arrow[r, "m"', tail]          & B                                  
    \end{tikzcd}
    \]
    By $(1)$ we have $md = 1_B$. Furthermore $mdm = m$ and, since $m$ is mono, we have that $dm = 1_C$. Thus $m$ is an iso.

        $(2 \implies 1)$ We want to prove that $f$ is a strong epi. Consider the pullback
        \[
        % https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBoAmAXVJADcBDAGwFcYkQBhEAX1PU1z5CKAMwVqdJq3YARHnxAZseAkTHEJDFm0QgAQvP7KhRMhppbpujgB0beALbwA+jIO8jg1SgAMpH5pSOiAAgjwSMFAA5vBEoABmAE4QDkhiIDgQSH6S2ux2DvQ4ABaJDsBoyQBW3M7EhiBJKdk0mUhkGfRYjOw4XT0WQeypNIz0AEYwjAAKAirCIIlYUcU4DU2piOltiOSDeboFRaXllRA1zuTryZt7GVmIHZbBtNfNiAAsrQ-pjFhgwSg9DgxUib02X3uSDukzAUDSOWe7Hi4KQkJ2OVh8MQAFoRIihrpmOFuEA
\begin{tikzcd}
    A \arrow[rd, dashed] \arrow[rrrd, "f", bend left] \arrow[rdd, "u", bend right] &                                                                       &  &                  \\
                                                                                   & C\times_DB \arrow[d, "\mathrm{proj}_1"] \arrow[rr, "\mathrm{proj}_2"] &  & B \arrow[d, "v"] \\
                                                                                   & C \arrow[rr, "m"', tail]                                              &  & D               
    \end{tikzcd}
    \]
Since $m$ is a mono, the projection $\mathrm{proj}_2$ is a mono. Now $f$ factors through $\mathrm{proj}_2$, so by $(2)$ $\mathrm{proj}_2$ is an iso. We define $d=\mathrm{proj}_1 (\mathrm{proj}_2)^{-1}$, which is the morphism we are looking for.
    \end{proof}
\end{subbox}

\begin{mainbox}{}
    \begin{defi}
        Let $\mathcal{C}$ be a category and $f : A \to B$, $g : A' \to B$ be morphisms. We say that the pair $(f,g)$ is a \textbf{strongly epimorphic pair} if for each commutative diagram of solid arrows
    \[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBpiBdUkANwEMAbAVxiRAEEQBfU9TXfIRQAGclVqMWbdgHJuvEBmx4CRAMxjq9Zq0QgAIvL7LBRAEykz47VL0BhI4v4qhyC8OuTdIAELdxMFAA5vBEoABmAE4QALZIZCA4EEgALFpebEGOUbFIoonJiGkSOmzhINQMdABGMAwACs6mepFYQQAWONnRcUXUSUgWIFW1DU2qegww4V3ppXrt3bmIGgWD-XRYDGw4m9tztiBxlTV1jSYTIK0dXTwRPXn9has23uFydyA5vQkDKwfeIJyE6jc4CS7XTpLXrFP6rBhYMDeKB0ODtQIVEqHAA62JgAA8sHAcHAAAQAQlJUH8XCAA
\begin{tikzcd}
    &                                     & B \arrow[rd, "h"] \arrow[dd, "\exists ! d", dashed] &   \\
A' \arrow[rru, "g"] \arrow[rrd, "g'"'] & A \arrow[ru, "f"'] \arrow[rd, "f'"] &                                                     & D \\
    &                                     & C \arrow[ru, "m"', tail]                            &  
\end{tikzcd}
    \]
    where $m$ is a mono, there exists a unique morphism $d$ such that $f'=df$, $g'=dg$ and $md=h$.
    \end{defi}
\end{mainbox}

\begin{mainbox}{}
    \begin{prop}\label{Eqivalence strong epi pair with iso}
    Let $\mathcal{C}$ be a category with pullbacks and $f : A \to B$, $g : A' \to B$ be morphisms. The following are equivalent:
    \begin{enumerate}
        \item $(f,g)$ is a strong epimorphic pair;
        \item Given factorizations $f=mf'$ and $g=mg'$, if $m$ is a mono, then it is an iso.
    \end{enumerate}
    \end{prop}
\end{mainbox}

\begin{subbox}{}
    \begin{proof}
        By adding the morphism $g'$ and its conditions in the proof of \Cref{Eqivalence strong epi with iso}, we obtain the proof of the current proposition.
\end{proof}
\end{subbox}

\begin{mainbox}{}
    \begin{prop}\label{Equivalence strong epi with coprod}
    Let $\mathcal{C}$ be a category with binary coproducts and $f : A \to B$, $g : A' \to B$ be morphisms. The following are equivalent:
    \begin{enumerate}
        \item $(f,g)$ is a strongly epimorphic pair;
        \item $[f,g]$ is a strong epimorphism.
    \end{enumerate}
    \end{prop}
\end{mainbox}

\begin{subbox}{}
    \begin{proof}
        $(1 \implies 2)$ $(f,g)$ is a strongly epimorphic pair. Consider the commutative diagram where $m$ is a mono
        \[ 
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAEEBqdgchAF9S6TLnyEUZAIxVajFmwDCAoSAzY8BIpNLTq9Zq0QgAIsuHqxW8jP3yjAIQEyYUAObwioAGYAnCAFskMhAcCCQAZj05QxBkL1JXCjMQXwCkbRCwxAAmahw6LAY2fMKQKIM2QMFvP0DEYND08rsQIuoGOgAjGAYABRENcRAfLFcACxxk1LrIzKRc2QqjMad+IA
\begin{tikzcd}
    A+A' \arrow[r, "{[f,g]}"] \arrow[d, "l"'] & B \arrow[d, "h"] \\
    C \arrow[r, "m", tail]                    & D               
    \end{tikzcd}
        \]
        Composing with the injection $\iota_1 : A \to A+A'$ and $\iota_2 : A' \to A+A'$, we have the following commutative diagram
        \[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBpiBdUkANwEMAbAVxiRAEEQBfU9TXfIRQAGclVqMWbdgHJuvEBmx4CRAMxjq9Zq0QgAIvL7LBRAEykz47VL0BhI4v4qhyC8OuTdIAELdxMFAA5vBEoABmAE4QALZIZCA4EEgALFpebEGOUbFIoonJiGkSOmzhINQMdABGMAwACs6mepFYQQAWONnRcUXUSUgWIFW1DU2qegww4V3ppXrt3bmIGgWD-XRYDGw4m9tztiBxlTV1jSYTIK0dXTwRPXn9has23gwAOu-4uwD6xEu9BIDFYHN6fb50H5mCrDU5jC5CK5tTr+LhAA
\begin{tikzcd}
    &                                           & B \arrow[rd, "h"]        &   \\
A' \arrow[rru, "g"] \arrow[rrd, "l\iota_2"'] & A \arrow[ru, "f"'] \arrow[rd, "l\iota_1"] &                          & D \\
    &                                           & C \arrow[ru, "m"', tail] &  
\end{tikzcd}
    \]
    Since $(f,g)$ is strongly epimorphic, there exists a unique morphism $d: B \to C$ such that $l\iota_1=df$, $l\iota_2=dg$ and $md=h$. The first two yield the equalities $d[f,g]\iota_1=df=l\iota_1$ and $[f,g]\iota_2=dg=l\iota_2$. So, by the universal property of the coproduct, we have that $d[f,g]=l$.
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAEEQBfU9TXfIRRkATFVqMWbdgHJuvEBmx4CRMgEZx9Zq0QcA1LPl9lgoutKbq2qXoBCxxfxVDkIy1sm6QAYUdKBVRQAFnJPHWl-ZzMQ0jFrL2k5HhNA11CrCQi9dkNkhQCXIgBWDwTs325xGCgAc3giUAAzACcIAFskMhAcCCR3LNsQAB1h-Bw6AH11R1aOpAsevsQBm29R8amRWbbOlepepABmcqHkJtJaih35xBOlpFDB7ygbvdKHxAB2U-WxiAm0zeSAAbAdlj9nmwNgCtsDvuCkAAOX5sBhVLhAA
\begin{tikzcd}
    A \arrow[d, "\iota_1"]    &                  &   &  & A \arrow[d, "\iota_1"]  &   \\
    A+A' \arrow[r, "{[f,g]}"] & B \arrow[r, "d"] & C &  & A+A' \arrow[r, "l"]     & C \\
    A' \arrow[u, "\iota_2"]   &                  &   &  & A' \arrow[u, "\iota_2"] &  
    \end{tikzcd}
\]
We already know that $md=h$, hence $[f,g]$ is a strong epimorphism.

$(2 \implies 1)$ $[f,g]$ is a strong epimorphism. Consider the following commutative diagram where $m$ is a mono
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZARgBpiBdUkANwEMAbAVxiRAEEQBfU9TXfIRQAGclVqMWbdgHJuvEBmx4CRAMxjq9Zq0QgAIvL7LBRAEykz47VL0BhI4v4qhyC8OuTdIAELdxMFAA5vBEoABmAE4QALZIZCA4EEgALFpebEGOUbFIoonJiGkSOmzhINQMdABGMAwACs6mepFYQQAWONnRcUXUSUgWIFW1DU2qegww4V3ppXrt3bmIGgWD-XRYDGw4m9tztiBxlTV1jSYTIK0dXTwRPXn9has23uFydyA5vQkDKwfeIJyE6jc4CS7XTr+LhAA
\begin{tikzcd}
    &                                     & B \arrow[rd, "h"]        &   \\
A' \arrow[rru, "g"] \arrow[rrd, "g'"'] & A \arrow[ru, "f"'] \arrow[rd, "f'"] &                          & D \\
    &                                     & C \arrow[ru, "m"', tail] &  
\end{tikzcd}
\]
Thus we have the commutative diagram
\[
% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAEEBqdgchAF9S6TLnyEUARnJVajFmwBCAoSAzY8BImQkz6zVohABhZcPViiUndT3zDAEQEyYUAObwioAGYAnCAFskMhAcCCQpWX02ZC9SVwpTEF8AoOpQpAAmGzkDEBieOJ4E6gY6ACMYBgAFEQ1xEB8sVwALHETkwMQI9MQAZmyow2b2v06skLC+gbsQQLS6LAY2HAWl-gp+IA
\begin{tikzcd}
    A+A' \arrow[r, "{[f,g]}"] \arrow[d, "{[f',g']}"'] & B \arrow[d, "h"] \\
    C \arrow[r, "m", tail]                            & D               
    \end{tikzcd}
\]
Since $[f,g]$ is a strong epimorphism, there exists a unique morphism $d: B \to C$ such that $d[f,g]=[f'.g']$ and $md=h$. Hence $f'=[f',g']\iota_1=d[f,g]\iota_1=df$ and $g'=[f',g']\iota_2=d[f,g]\iota_2=dg$. Thus $(f,g)$ is a strongly epimorphic pair.
    \end{proof}
\end{subbox}
